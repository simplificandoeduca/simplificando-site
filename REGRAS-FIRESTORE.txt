rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function logado() {
      return request.auth != null;
    }
    // dono do painel: a primeira conta que entrou no painel
    function dono() {
      return logado()
        && get(/databases/$(database)/documents/sistema/dono).data.uid == request.auth.uid;
    }

    // quem é o dono (o painel consulta antes do login para saber se é o primeiro acesso)
    match /sistema/dono {
      allow get: if true;
      allow create: if logado()
        && !exists(/databases/$(database)/documents/sistema/dono)
        && request.resource.data.uid == request.auth.uid;
    }

    // dados do painel: só o personal
    match /alunos/{id}     { allow read, write: if dono(); }
    match /exercicios/{id} { allow read, write: if dono(); }
    match /modelos/{id}    { allow read, write: if dono(); }
    match /config/{id}     { allow read, write: if dono(); }

    // o que o app do aluno lê: cada aluno só o seu, e só com o acesso liberado
    match /appAlunos/{uid} {
      allow get: if logado() && request.auth.uid == uid && resource.data.ativo == true;
      allow read, write: if dono();
    }

    // resumo de cada conta (nome, tipo, último acesso, aceite dos termos) para a aba Perfis do painel;
    // sem dados de saúde. Cada um grava o seu; só o dono do painel lê todos
    match /perfis/{uid} {
      allow read: if dono() || (logado() && request.auth.uid == uid);
      allow create, update: if logado() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['tipo', 'aid', 'nome', 'email', 'sexo', 'objetivo', 'plano', 'treinos', 'criadoEm', 'ultimoAcesso', 'aceite', 'acomp', 'ver'])
        && request.resource.data.tipo in ['autonomo', 'personal'];
      allow delete: if dono() || (logado() && request.auth.uid == uid);
    }

    // aluno que treina sem personal: cada um lê e grava só os próprios treinos;
    // o dono do painel só lê se a pessoa permitiu o acompanhamento no app (perfis/{uid}.acomp == true)
    match /autonomos/{uid} {
      allow read, delete: if logado() && request.auth.uid == uid;
      allow get: if dono()
        && exists(/databases/$(database)/documents/perfis/$(uid))
        && get(/databases/$(database)/documents/perfis/$(uid)).data.get('acomp', false) == true;
      allow create, update: if logado() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['s', 'em'])
        && request.resource.data.s is string
        && request.resource.data.s.size() < 900000;
    }

    // o que o aluno com personal fez no app (treinos, cargas, academia, comentários, cardio, água e check-in):
    // cada aluno grava só o seu, e só se o acesso dele estiver liberado no painel; só o dono do painel lê todos
    match /progresso/{uid} {
      allow read: if dono() || (logado() && request.auth.uid == uid);
      allow create, update: if logado() && request.auth.uid == uid
        && exists(/databases/$(database)/documents/appAlunos/$(request.auth.uid))
        && get(/databases/$(database)/documents/appAlunos/$(request.auth.uid)).data.aid == request.resource.data.aid
        && get(/databases/$(database)/documents/appAlunos/$(request.auth.uid)).data.ativo == true
        && request.resource.data.keys().hasOnly(['aid', 's', 'em'])
        && request.resource.data.s is string
        && request.resource.data.s.size() < 900000;
      allow delete: if dono() || (logado() && request.auth.uid == uid);
      // o painel passa os treinos para a conta nova quando o aluno muda para o login com e-mail
      allow write: if dono();
    }

    // senha de 6 números que o aluno com personal escolheu pelo e-mail de "esqueci a senha":
    // o aluno grava só a sua; o dono do painel lê para mostrar o código atualizado
    match /codigos/{uid} {
      allow read: if dono() || (logado() && request.auth.uid == uid);
      allow create, update: if logado() && request.auth.uid == uid
        && exists(/databases/$(database)/documents/appAlunos/$(request.auth.uid))
        && request.resource.data.keys().hasOnly(['codigo', 'em'])
        && request.resource.data.codigo is string
        && request.resource.data.codigo.matches('^[0-9]{6}$')
        && request.resource.data.em is int;
      allow delete: if dono();
    }

    // finanças pessoais do dono do painel (pasta /financas): só ele lê e grava, um documento só
    match /financas/{uid} {
      allow read, delete: if dono() && request.auth.uid == uid;
      allow create, update: if dono() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['s', 'em'])
        && request.resource.data.s is string
        && request.resource.data.s.size() < 900000
        && request.resource.data.em is int;
    }
  }
}
